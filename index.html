Skema SQL Tabel Supabase SIDAK ("sidak_data_utama")

Salin dan jalankan di SQL Editor proyek Supabase SIDAK Anda nanti.

CREATE TABLE IF NOT EXISTS public.sidak_data_utama (
  id TEXT PRIMARY KEY,
  tipe_data TEXT NOT NULL CHECK (tipe_data IN ('sahabat_dakwah', 'mitra_dakwah', 'masjid_binaan', 'dpc', 'kader_master', 'khutbah_jumat', 'taklim')),
  payload JSONB NOT NULL,
  updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

ALTER TABLE public.sidak_data_utama ENABLE ROW LEVEL SECURITY;

CREATE POLICY "Akses SIDAK Pasangkayu" ON public.sidak_data_utama
  FOR ALL USING (true) WITH CHECK (true);
